# Mailwarden
#
# The marketing pages are crawlable. Everything behind sign-in is not:
# /app and /api return either a redirect or a 401 to a crawler, so indexing
# them wastes crawl budget and can surface a login wall as a search result for
# the product's own name.
#
# This file is not a security control — /admin.html is admin-gated server-side
# and carries <meta name="robots" content="noindex"> of its own. Listing a
# private path in robots.txt is what tells a scanner where to look, so the
# blanket /api and /auth rules deliberately do not enumerate anything.

User-agent: *
Allow: /$
Allow: /try.html
Allow: /pricing.html
Allow: /privacy.html
Allow: /terms.html
Allow: /blog/
# /try.html renders its inbox from this one endpoint. It is public, static and
# cacheable, and a crawler that cannot fetch it sees an empty page where the
# demo should be. Longest-match wins, so this survives the /api/ rule below.
Allow: /api/demo/inbox
Disallow: /app
Disallow: /api/
Disallow: /auth/
Disallow: /admin.html

Sitemap: https://mailwarden.xyz/sitemap.xml
