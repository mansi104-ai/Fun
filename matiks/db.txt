model user{
  id
  email
  password
  username

  gamesMembers.   GameMembers[]
  friends.   Friends[]

}

model game{
  id
  timelimit (60 seconds)
  startTime (date)
  questions Question[]
  questionAnswer[]
  members. Members[]

}

model GameMember{
  id
  userId
  gameId

}

  senderId
  receiverId
  status : *Accepted*| *Rejected* | *pending*
}

    operation 2 : number 
    sign: "+"|"-" | "*" | "/"
    gameId : string

}

model questionAnswer{
  id
  questionId
  gameId
}
